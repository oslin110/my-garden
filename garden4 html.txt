<script>
    // 切換第一層大選項（花園 / 廚房）
    function switchMainTab(sectionId, btn, activeClass) {
        document.querySelectorAll('.section-content').forEach(s => s.classList.remove('active'));
        document.querySelectorAll('.main-tab-btn').forEach(b => {
            b.classList.remove('active-garden');
            b.classList.remove('active-kitchen');
        });

        document.getElementById(sectionId).classList.add('active');
        btn.classList.add(activeClass);
    }

    // 切換第二層項目（植物或食譜卡片）
    function switchItem(sectionId, itemId, btn) {
        const section = document.getElementById(sectionId);
        section.querySelectorAll('.item-card').forEach(c => c.classList.remove('active'));
        section.querySelectorAll('.menu-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(itemId).classList.add('active');
        btn.classList.add('active');
    }

    // 切換相簿照片
    function showImg(cardId, index) {
        const card = document.getElementById(cardId);
        const imgs = card.querySelectorAll('.image-container img');
        const btns = card.querySelectorAll('.nav-btn');
        imgs.forEach((img, i) => img.classList.toggle('active', i === index));
        btns.forEach((btn, i) => btn.classList.toggle('active', i === index));
    }
</script>
